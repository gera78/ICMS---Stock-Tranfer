*&---------------------------------------------------------------------*
*& Include Y0033FI_ICMS_F03
*&---------------------------------------------------------------------*
*& CR 16002 - API for a single nota fiscal (DOCNUM). Included ONLY by
*& function group Y0033FI_ICMS_API (see LY0033FI_ICMS_APITOP), which is
*& called by class Y0033CL_ICMS_STOCK_TRANSFER - never by the two
*& report programs.
*&
*& Reuses the report programs' own FORMs (TOP/F01/F02) unchanged, so a
*& DOCNUM processed here goes through exactly the same search, credit
*& calculation, auto-approval, posting and log as in the Admin program:
*&
*&   API_AUTO_POST          - search + save as 021 (Auto Approved) +
*&                            post now, same as P_AUTO + Create
*&                            Document(s), for this DOCNUM only.
*&   API_GET_POSTING_DATA   - same search/calculation, but instead of
*&                            posting, returns the BAPI_ACC_DOCUMENT_POST
*&                            GL/currency lines so another process can
*&                            add them to its own accounting document.
*&                            Writes nothing to Y0033FI_ICM_ITEM.
*&   API_CONFIRM_POSTING    - after that other document is posted: saves
*&                            the items (021) and marks them 040 - Posted
*&                            with the caller's BELNR/GJAHR, with log.
*&
*& Which search model runs is decided by the DOCNUM itself:
*&   J_1BNFDOC-DIRECT = '2' (origin/outgoing) -> stock-transfer model
*&                                               (SELECT_TRANSFER_CANDIDATES)
*&   J_1BNFDOC-DIRECT = '1' (destination/incoming) -> eDocument model
*&                                               (SELECT_CANDIDATE_DOCUMENTS),
*&                            restricted to that nota's eDocuments via
*&                            GR_EDOC_GUID.
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*&      Form  API_PREPARE
*&---------------------------------------------------------------------*
* Resets every global the shared FORMs read (the selection-screen
* substitutes declared in LY0033FI_ICMS_APITOP, GIT_ITEM, GR_EDOC_GUID),
* validates the DOCNUM and loads the company parameters. PC_OK =
* abap_false means stop - the reason is already in PC_RETURN.
*---------------------------------------------------------------------*
FORM api_prepare USING    pu_bukrs  TYPE bukrs
                          pu_docnum TYPE j_1bdocnum
                 CHANGING pc_ok     TYPE abap_bool
                          pc_return TYPE bapiret2_t.

  pc_ok = abap_false.
  gv_api_mode = abap_true.
  CLEAR: git_item, gr_edoc_guid,
         s_datum, s_nfenum, s_docnum, s_series, s_docdat, s_autdat, s_acckey, s_lcto, s_actdat,
         p_mode1, p_mode2.
  " Action-date filter checkboxes (REFRESH_FROM_DB) - not used by the
  " API, set like the report's defaults only for consistency.
  p_asav = p_aapr = p_arej = p_apst = abap_true.

  " Same authorization as CHECK_BUKRS_AUTHORITY, but returned instead
  " of a TYPE E message (which would abort the caller).
  AUTHORITY-CHECK OBJECT 'F_BKPF_BUK'
    ID 'BUKRS' FIELD pu_bukrs
    ID 'ACTVT' FIELD '03'.
  IF sy-subrc <> 0.
    APPEND VALUE #( type = 'E' message = |No FI authorization for company code { pu_bukrs }| ) TO pc_return.
    RETURN.
  ENDIF.

  SELECT SINGLE docnum, bukrs, direct, cancel, nfenum, series
    FROM j_1bnfdoc
    INTO @DATA(ls_doc)
    WHERE docnum = @pu_docnum.
  IF sy-subrc <> 0.
    APPEND VALUE #( type = 'E' message = |Nota fiscal { pu_docnum } not found| ) TO pc_return.
    RETURN.
  ENDIF.
  IF ls_doc-bukrs <> pu_bukrs.
    APPEND VALUE #( type = 'E' message = |Nota fiscal { pu_docnum } belongs to company code { ls_doc-bukrs }, not { pu_bukrs }| ) TO pc_return.
    RETURN.
  ENDIF.
  IF ls_doc-cancel = abap_true.
    APPEND VALUE #( type = 'E' message = |Nota fiscal { pu_docnum } is cancelled| ) TO pc_return.
    RETURN.
  ENDIF.

  PERFORM get_company_currency USING pu_bukrs CHANGING gv_waers.
  PERFORM get_params USING pu_bukrs.
  gv_id_exec = sy-uname && sy-datum && sy-uzeit.  " same format as INITIALIZATION in the report programs

  APPEND VALUE #( sign = 'I' option = 'EQ' low = pu_docnum ) TO s_docnum.

  CASE ls_doc-direct.
    WHEN '2'.
      p_mode2 = abap_true.  " origin nota - stock-transfer model

    WHEN '1'.
      p_mode1 = abap_true.  " destination nota - eDocument model
      " Restrict the EDOCUMENT read to this nota's eDocuments - without
      " it, SELECT_CANDIDATE_DOCUMENTS would read every completed
      " eDocument of the type (no date filter here) just to keep one.
      " Sources: eDocuments already linked by a previous search
      " (Y0033FI_ICM_EDOC), plus EDOBRINCOMING by NFENUM+SERIES for one
      " never seen yet. The cardinality/lane rules inside
      " SELECT_CANDIDATE_DOCUMENTS still decide as usual.
      SELECT edoc_guid FROM y0033fi_icm_edoc
        INTO TABLE @DATA(lt_known)
        WHERE docnum = @pu_docnum
          AND status <> @gc_edoc_status-link_error.
      LOOP AT lt_known INTO DATA(lv_known).
        APPEND VALUE #( sign = 'I' option = 'EQ' low = lv_known ) TO gr_edoc_guid.
      ENDLOOP.

      " EDOBRINCOMING-NFENUM/SERIES are not the same data elements as
      " J_1BNFDOC's (see SELECT_CANDIDATE_DOCUMENTS) - moved into
      " variables of EDOBRINCOMING's own types before the WHERE.
      DATA lv_inc_nfenum TYPE edobrincoming-nfenum.
      DATA lv_inc_series TYPE edobrincoming-series.
      lv_inc_nfenum = ls_doc-nfenum.
      lv_inc_series = ls_doc-series.
      SELECT edoc_guid FROM edobrincoming
        INTO TABLE @DATA(lt_inc_guid)
        WHERE nfenum = @lv_inc_nfenum
          AND series = @lv_inc_series.
      LOOP AT lt_inc_guid INTO DATA(lv_inc_guid).
        APPEND VALUE #( sign = 'I' option = 'EQ' low = lv_inc_guid ) TO gr_edoc_guid.
      ENDLOOP.

      SORT gr_edoc_guid BY low.
      DELETE ADJACENT DUPLICATES FROM gr_edoc_guid COMPARING low.
      IF gr_edoc_guid IS INITIAL.
        APPEND VALUE #( type = 'E' message = |No eDocument found for nota fiscal { pu_docnum }| ) TO pc_return.
        RETURN.
      ENDIF.

    WHEN OTHERS.
      APPEND VALUE #( type = 'E' message = |Nota fiscal { pu_docnum } has unexpected direction { ls_doc-direct }| ) TO pc_return.
      RETURN.
  ENDCASE.

  pc_ok = abap_true.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  API_FINISH
*&---------------------------------------------------------------------*
* Leaves the function group's globals clean for the next call in the
* same session.
*---------------------------------------------------------------------*
FORM api_finish.
  gv_api_mode = abap_false.
  CLEAR: git_item, gr_edoc_guid, s_docnum, p_mode1, p_mode2.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  API_AUTO_POST
*&---------------------------------------------------------------------*
* Same as the Admin program with P_AUTO + Create Document(s), for one
* DOCNUM: new candidates are saved as 021 - Auto Approved (with log),
* then every item of this DOCNUM waiting to be posted (020, 021, 025)
* is posted right away - one accounting document, with the same
* cancelled-nota check and immediate retry on 025 as the program.
* Commits its own work (standalone call, not inside another LUW).
*
* PC_ITEMS: every Y0033FI_ICM_ITEM row of this DOCNUM after the run.
* PC_RETURN: one message per item processed - S = posted (BELNR/GJAHR),
* E = failed (reason), W = skipped/nothing to do.
*---------------------------------------------------------------------*
FORM api_auto_post USING    pu_bukrs  TYPE bukrs
                            pu_docnum TYPE j_1bdocnum
                   CHANGING pc_items  TYPE y0033fi_icm_item_tt
                            pc_return TYPE bapiret2_t.

  CLEAR: pc_items, pc_return.

  DATA lv_ok TYPE abap_bool.
  PERFORM api_prepare USING pu_bukrs pu_docnum CHANGING lv_ok pc_return.
  IF lv_ok = abap_false.
    PERFORM api_finish.
    RETURN.
  ENDIF.

  " 1) New candidates -> 021 (same FORM as P_AUTO; commits)
  PERFORM auto_approve_candidates USING pu_bukrs gv_id_exec.

  " 2) Everything of this DOCNUM waiting to be posted
  DATA lt_post TYPE y0033fi_icm_item_tt.
  SELECT * FROM y0033fi_icm_item
    INTO TABLE @lt_post
    WHERE docnum = @pu_docnum
      AND ( status = @gc_status-approved
         OR status = @gc_status-auto_approved
         OR status = @gc_status-post_failed ).

  IF lt_post IS INITIAL.
    APPEND VALUE #( type = 'W' message = |Nothing to post for nota fiscal { pu_docnum } (no new candidate, no item waiting)| ) TO pc_return.
  ELSE.
    DATA(lt_sent) = lt_post.
    PERFORM post_item_list CHANGING lt_post.

    " 3) Outcome per item, re-read from the table (POST_ONE_DOCUMENT
    " reports back only through Y0033FI_ICM_ITEM, same as in
    " POST_RELEASED_ITEMS).
    LOOP AT lt_sent INTO DATA(ls_sent).
      SELECT SINGLE * FROM y0033fi_icm_item
        INTO @DATA(ls_now)
        WHERE docnum = @ls_sent-docnum AND itmnum = @ls_sent-itmnum AND seqnr = @ls_sent-seqnr.
      CASE ls_now-status.
        WHEN gc_status-posted.
          APPEND VALUE #( type = 'S' message = |Item { ls_now-itmnum } posted: { ls_now-belnr_fi }/{ ls_now-gjahr_fi }| ) TO pc_return.
        WHEN gc_status-cancelled_pre_post.
          APPEND VALUE #( type = 'W' message = |Item { ls_now-itmnum } skipped: nota fiscal cancelled| ) TO pc_return.
        WHEN gc_status-posting.
          APPEND VALUE #( type = 'E' message = |Item { ls_now-itmnum } left in 030 - Posting for manual review| ) TO pc_return.
        WHEN OTHERS.
          APPEND VALUE #( type = 'E' message = |Item { ls_now-itmnum } not posted ({ ls_now-status }): { ls_now-msg }| ) TO pc_return.
      ENDCASE.
    ENDLOOP.
  ENDIF.

  SELECT * FROM y0033fi_icm_item
    INTO TABLE @pc_items
    WHERE docnum = @pu_docnum.

  PERFORM api_finish.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  API_GET_POSTING_DATA
*&---------------------------------------------------------------------*
* Same items API_AUTO_POST would post for this DOCNUM - new candidates
* (calculated now, NOT saved) plus items already waiting (020, 021,
* 025) - returned as BAPI_ACC_DOCUMENT_POST lines instead of posted,
* so another process can add them to its own accounting document.
* Nothing is written to Y0033FI_ICM_ITEM and nothing is committed here:
* after the caller's document is posted, it must call
* API_CONFIRM_POSTING with PC_ITEMS as returned (that is what saves the
* items and marks them 040 - otherwise the job would post them again).
*
* PU_ITEMNO_OFFSET: ITEMNO_ACC of the returned lines starts at
* offset + 1, so they can be appended after the caller's own lines.
* PC_ITEMS: the items behind the lines; BUZEI_FI = ITEMNO_ACC of each
* item's debit line, SEQNR = 000 for new (not yet saved) items.
*---------------------------------------------------------------------*
FORM api_get_posting_data USING    pu_bukrs         TYPE bukrs
                                   pu_docnum        TYPE j_1bdocnum
                                   pu_itemno_offset TYPE posnr_acc
                          CHANGING pc_gl            TYPE ty_it_acc_gl
                                   pc_cur           TYPE ty_it_acc_cur
                                   pc_items         TYPE y0033fi_icm_item_tt
                                   pc_return        TYPE bapiret2_t.

  CLEAR: pc_gl, pc_cur, pc_items, pc_return.

  DATA lv_ok TYPE abap_bool.
  PERFORM api_prepare USING pu_bukrs pu_docnum CHANGING lv_ok pc_return.
  IF lv_ok = abap_false.
    PERFORM api_finish.
    RETURN.
  ENDIF.

  " 1) Items already saved and waiting to be posted
  DATA lt_items TYPE y0033fi_icm_item_tt.
  SELECT * FROM y0033fi_icm_item
    INTO TABLE @lt_items
    WHERE docnum = @pu_docnum
      AND ( status = @gc_status-approved
         OR status = @gc_status-auto_approved
         OR status = @gc_status-post_failed ).

  " 2) New candidates - same chain as "Search New Candidates"; items
  " with an active cycle (step 1 included) are dropped by
  " FILTER_ACTIVE_CYCLE, so nothing is returned twice.
  IF p_mode1 = abap_true.
    PERFORM select_candidate_documents USING pu_bukrs.
  ENDIF.
  IF p_mode2 = abap_true.
    PERFORM select_transfer_candidates USING pu_bukrs.
  ENDIF.
  PERFORM read_items_and_icms.
  PERFORM determine_benefit_percentage.
  PERFORM filter_active_cycle.
  PERFORM calculate_credit.
  PERFORM flag_not_candidates.

  LOOP AT git_item INTO gwa_item.
    CASE gwa_item-status.
      WHEN gc_status-ready_to_save.
        DATA ls_new TYPE y0033fi_icm_item.
        CLEAR ls_new.
        MOVE-CORRESPONDING gwa_item TO ls_new.
        ls_new-seqnr  = 0.                         " not saved yet - see API_CONFIRM_POSTING
        ls_new-status = gc_status-auto_approved.   " what it becomes when confirmed
        APPEND ls_new TO lt_items.
      WHEN gc_status-error.
        APPEND VALUE #( type = 'W' message = |Item { gwa_item-itmnum } not included (error): { gwa_item-msg }| ) TO pc_return.
      WHEN gc_status-not_candidate.
        APPEND VALUE #( type = 'I' message = |Item { gwa_item-itmnum } not included: { gwa_item-msg }| ) TO pc_return.
    ENDCASE.
  ENDLOOP.

  IF lt_items IS INITIAL.
    APPEND VALUE #( type = 'W' message = |No ICMS benefit to post for nota fiscal { pu_docnum }| ) TO pc_return.
    PERFORM api_finish.
    RETURN.
  ENDIF.

  SORT lt_items BY docnum itmnum seqnr.

  " 3) The same lines POST_ONE_DOCUMENT would post
  DATA lt_gl    TYPE ty_it_acc_gl.
  DATA lt_cur   TYPE ty_it_acc_cur.
  DATA lt_buzei TYPE ty_it_buzei_map.
  PERFORM build_posting_lines USING lt_items pu_itemno_offset
                              CHANGING lt_gl lt_cur lt_buzei.

  LOOP AT lt_items ASSIGNING FIELD-SYMBOL(<ls_item>).
    READ TABLE lt_buzei INTO DATA(ls_buzei)
      WITH KEY docnum = <ls_item>-docnum itmnum = <ls_item>-itmnum seqnr = <ls_item>-seqnr.
    IF sy-subrc = 0.
      <ls_item>-buzei_fi = ls_buzei-buzei.
    ENDIF.
  ENDLOOP.

  " BAPI_ACC_DOCUMENT_POST accepts at most 999 lines per document
  IF pu_itemno_offset + lines( lt_gl ) > 999.
    APPEND VALUE #( type = 'W' message = |{ lines( lt_gl ) } lines + offset { pu_itemno_offset } exceed 999 lines per document| ) TO pc_return.
  ENDIF.

  pc_gl    = lt_gl.
  pc_cur   = lt_cur.
  pc_items = lt_items.
  APPEND VALUE #( type = 'S' message = |{ lines( lt_items ) } item(s), { lines( lt_gl ) } line(s) for nota fiscal { pu_docnum }| ) TO pc_return.

  PERFORM api_finish.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  API_CONFIRM_POSTING
*&---------------------------------------------------------------------*
* Call right after the caller's accounting document (which included
* the lines from API_GET_POSTING_DATA) is posted. For each item in
* PU_ITEMS (as returned by API_GET_POSTING_DATA):
*   SEQNR = 000 (new) -> saved as 021 - Auto Approved (log
*                        AUTO_APPROVED), same checks as
*                        AUTO_APPROVE_CANDIDATES;
*   SEQNR > 000        -> must still be 020/021/025;
* then marked 040 - Posted with the caller's BELNR/GJAHR/dates and
* BUZEI_FI as returned (log POSTED, "posted within external document").
*
* PU_COMMIT = abap_false (default): no COMMIT WORK - the updates belong
* to the caller's LUW and are committed (or rolled back) together with
* its own document. abap_true: COMMIT WORK AND WAIT here.
*---------------------------------------------------------------------*
FORM api_confirm_posting USING    pu_bukrs  TYPE bukrs
                                  pu_docnum TYPE j_1bdocnum
                                  pu_belnr  TYPE belnr_d
                                  pu_gjahr  TYPE gjahr
                                  pu_budat  TYPE budat
                                  pu_bldat  TYPE bldat
                                  pu_items  TYPE y0033fi_icm_item_tt
                                  pu_commit TYPE abap_bool
                         CHANGING pc_return TYPE bapiret2_t.

  CLEAR pc_return.

  IF pu_belnr IS INITIAL OR pu_gjahr IS INITIAL.
    APPEND VALUE #( type = 'E' message = 'Accounting document number/year missing - nothing confirmed' ) TO pc_return.
    RETURN.
  ENDIF.

  gv_api_mode = abap_true.
  gv_id_exec = sy-uname && sy-datum && sy-uzeit.

  DATA lv_msg TYPE char200.
  lv_msg = |Posted within external accounting document { pu_belnr }/{ pu_gjahr }|.

  LOOP AT pu_items INTO DATA(ls_it) WHERE docnum = pu_docnum AND bukrs = pu_bukrs.
    DATA lv_subrc TYPE sy-subrc.
    DATA lv_lock_seqnr TYPE y0033fi_icm_seq.
    lv_lock_seqnr = ls_it-seqnr.
    PERFORM enqueue_item USING ls_it-docnum ls_it-itmnum lv_lock_seqnr CHANGING lv_subrc.
    IF lv_subrc <> 0.
      APPEND VALUE #( type = 'E' message = |Item { ls_it-itmnum } locked by another user - not confirmed| ) TO pc_return.
      CONTINUE.
    ENDIF.

    DATA lv_from TYPE y0033fi_icm_status_n.

    IF ls_it-seqnr IS INITIAL.
      " New item - same race protection as AUTO_APPROVE_CANDIDATES
      SELECT COUNT(*) FROM y0033fi_icm_item
        INTO @DATA(lv_active_count)
        WHERE docnum = @ls_it-docnum
          AND itmnum = @ls_it-itmnum
          AND status <> @gc_status-reversed
          AND status <> @gc_status-rejected
          AND status <> @gc_status-cancelled_pre_post.
      IF lv_active_count > 0.
        APPEND VALUE #( type = 'E' message = |Item { ls_it-itmnum } already has an active cycle - check for a duplicated benefit in { pu_belnr }| ) TO pc_return.
        PERFORM dequeue_item USING ls_it-docnum ls_it-itmnum lv_lock_seqnr.
        CONTINUE.
      ENDIF.

      DATA lv_seqnr TYPE n LENGTH 3.
      PERFORM next_seqnr USING ls_it-docnum ls_it-itmnum CHANGING lv_seqnr.
      ls_it-seqnr    = lv_seqnr.
      ls_it-status   = gc_status-auto_approved.
      ls_it-cre_user = sy-uname.
      ls_it-cre_date = sy-datum.
      ls_it-cre_time = sy-uzeit.
      ls_it-apr_user = sy-uname.
      ls_it-apr_date = sy-datum.
      ls_it-apr_time = sy-uzeit.
      ls_it-id_exec  = gv_id_exec.
      INSERT y0033fi_icm_item FROM @ls_it.
      IF sy-subrc <> 0.
        APPEND VALUE #( type = 'E' message = |Item { ls_it-itmnum } could not be saved - not confirmed| ) TO pc_return.
        PERFORM dequeue_item USING ls_it-docnum ls_it-itmnum lv_lock_seqnr.
        CONTINUE.
      ENDIF.
      PERFORM write_log USING ls_it gc_action-auto_approved '000' gc_status-auto_approved
                               'Candidate auto-approved (external posting)'.
      IF ls_it-edoc_guid IS NOT INITIAL.
        UPDATE y0033fi_icm_edoc SET seqnr = ls_it-seqnr
          WHERE docnum = ls_it-docnum AND edoc_guid = ls_it-edoc_guid.
      ENDIF.
      lv_from = gc_status-auto_approved.
    ELSE.
      SELECT SINGLE status FROM y0033fi_icm_item
        INTO @lv_from
        WHERE docnum = @ls_it-docnum AND itmnum = @ls_it-itmnum AND seqnr = @ls_it-seqnr.
      IF sy-subrc <> 0
         OR ( lv_from <> gc_status-approved AND lv_from <> gc_status-auto_approved
              AND lv_from <> gc_status-post_failed ).
        APPEND VALUE #( type = 'E' message = |Item { ls_it-itmnum }/{ ls_it-seqnr } is in status { lv_from } - not confirmed| ) TO pc_return.
        PERFORM dequeue_item USING ls_it-docnum ls_it-itmnum lv_lock_seqnr.
        CONTINUE.
      ENDIF.
    ENDIF.

    " STATUS = LV_FROM guards against a parallel run that moved the item
    " on since it was read (same pattern as APPROVE_CANDIDATES).
    UPDATE y0033fi_icm_item SET status   = @gc_status-posted,
                                belnr_fi = @pu_belnr,
                                gjahr_fi = @pu_gjahr,
                                buzei_fi = @ls_it-buzei_fi,
                                bldat_fi = @pu_bldat,
                                budat_fi = @pu_budat,
                                period_fallback = @abap_false,
                                pst_user = @sy-uname,
                                pst_date = @sy-datum,
                                pst_time = @sy-uzeit,
                                msg      = @space
      WHERE docnum = @ls_it-docnum AND itmnum = @ls_it-itmnum AND seqnr = @ls_it-seqnr
        AND status = @lv_from.
    IF sy-subrc = 0.
      ls_it-belnr_fi = pu_belnr.
      ls_it-gjahr_fi = pu_gjahr.
      ls_it-budat_fi = pu_budat.
      ls_it-bldat_fi = pu_bldat.
      PERFORM write_log USING ls_it gc_action-posted lv_from gc_status-posted lv_msg.
      APPEND VALUE #( type = 'S' message = |Item { ls_it-itmnum } confirmed as posted in { pu_belnr }/{ pu_gjahr }| ) TO pc_return.
    ELSE.
      APPEND VALUE #( type = 'E' message = |Item { ls_it-itmnum } changed by another process - not confirmed| ) TO pc_return.
    ENDIF.

    PERFORM dequeue_item USING ls_it-docnum ls_it-itmnum lv_lock_seqnr.
  ENDLOOP.

  IF pc_return IS INITIAL.
    APPEND VALUE #( type = 'W' message = |No item of nota fiscal { pu_docnum } to confirm| ) TO pc_return.
  ENDIF.

  IF pu_commit = abap_true.
    COMMIT WORK AND WAIT.
  ENDIF.

  gv_api_mode = abap_false.

ENDFORM.
