MODULE status_0400 OUTPUT.
  SET PF-STATUS 'STATUS_0400'.
  SET TITLEBAR 'TITLE_0400'.

  IF go_cc_param IS NOT BOUND.
    CREATE OBJECT go_cc_param EXPORTING container_name = 'CC_PARAM'.
    CREATE OBJECT go_alv_param EXPORTING i_parent = go_cc_param.
    PERFORM load_param_table.
    PERFORM build_param_fieldcat.
    go_alv_param->set_table_for_first_display(
      EXPORTING is_layout = VALUE lvc_s_layo( sel_mode = 'A' zebra = 'X' edit = 'X' cwidth_opt = 'X' )
      CHANGING it_outtab = git_param it_fieldcatalog = gt_fcat_param ).
  ELSE.
    go_alv_param->refresh_table_display( ).
  ENDIF.
ENDMODULE.

MODULE user_command_0400 INPUT.
  gv_ucomm = sy-ucomm.
  CLEAR sy-ucomm.

  CASE gv_ucomm.
    WHEN 'ADD_ROW'.
      go_alv_param->check_changed_data( ).
      APPEND VALUE y0033fi_icm_parm( ) TO git_param.
      go_alv_param->refresh_table_display( ).
    WHEN 'DEL_ROW'.
      go_alv_param->check_changed_data( ).
*      DATA lt_rows TYPE lvc_t_row.
      go_alv_param->get_selected_rows( IMPORTING et_index_rows = lt_rows ).
      SORT lt_rows BY index DESCENDING.
      LOOP AT lt_rows INTO ls_row.
        DELETE git_param INDEX ls_row-index.
      ENDLOOP.
      go_alv_param->refresh_table_display( ).
    WHEN 'SAVE'.
      go_alv_param->check_changed_data( ).
      PERFORM save_param_table.
      PERFORM load_param_table.
      go_alv_param->refresh_table_display( ).
    " &F03/&F12/&F15 are the raw system function codes some SAP GUI
    " toolbar icons send instead of BACK/EXIT/CANC - catching both.
    WHEN 'BACK' OR 'EXIT' OR 'CANC' OR 'CANCEL' OR '&F03' OR '&F12' OR '&F15'.
      go_cc_param->free( ).
      CLEAR go_cc_param.
      CLEAR go_alv_param.
      LEAVE TO SCREEN 0.
  ENDCASE.
ENDMODULE.
