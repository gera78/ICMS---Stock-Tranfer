*&---------------------------------------------------------------------*
*& Include Y0033FI_ICMS_SCR_SELPARM
*&---------------------------------------------------------------------*
*& CR 16002 - PBO/PAI modules for screen 500 ("Maintain Selection
*& Criteria", editable ALV on Y0033FI_ICM_SELPARM). Included only by
*& Y0033FI_ICM_STOCK_TRANSFER_ADM, after Y0033FI_ICMS_F02 (the forms
*& called here - LOAD_SELPARM_TABLE, BUILD_SELPARM_FIELDCAT,
*& SAVE_SELPARM_TABLE - live there).
*&---------------------------------------------------------------------*

MODULE status_0500 OUTPUT.
  SET PF-STATUS 'STATUS_0500'.
  SET TITLEBAR 'TITLE_0500'.

  IF go_cc_selparm IS NOT BOUND.
    CREATE OBJECT go_cc_selparm EXPORTING container_name = 'CC_SELPARM'.
    CREATE OBJECT go_alv_selparm EXPORTING i_parent = go_cc_selparm.
    PERFORM load_selparm_table.
    PERFORM build_selparm_fieldcat.
    go_alv_selparm->set_table_for_first_display(
      EXPORTING is_layout = VALUE lvc_s_layo( sel_mode = 'A' zebra = 'X' edit = 'X' cwidth_opt = 'X' )
      CHANGING it_outtab = git_selparm it_fieldcatalog = gt_fcat_selparm ).
  ELSE.
    go_alv_selparm->refresh_table_display( ).
  ENDIF.
ENDMODULE.

MODULE user_command_0500 INPUT.
  gv_ucomm = sy-ucomm.
  CLEAR sy-ucomm.

  CASE gv_ucomm.
    WHEN 'ADD_ROW'.
      go_alv_selparm->check_changed_data( ).
      APPEND VALUE y0033fi_icm_selparm( ) TO git_selparm.
      go_alv_selparm->refresh_table_display( ).
    WHEN 'DEL_ROW'.
      go_alv_selparm->check_changed_data( ).
      DATA lt_rows TYPE lvc_t_row.
      go_alv_selparm->get_selected_rows( IMPORTING et_index_rows = lt_rows ).
      SORT lt_rows BY index DESCENDING.
      LOOP AT lt_rows INTO DATA(ls_row).
        DELETE git_selparm INDEX ls_row-index.
      ENDLOOP.
      go_alv_selparm->refresh_table_display( ).
    WHEN 'SAVE'.
      go_alv_selparm->check_changed_data( ).
      PERFORM save_selparm_table.
      PERFORM load_selparm_table.
      go_alv_selparm->refresh_table_display( ).
    " &F03/&F12/&F15 are the raw system function codes some SAP GUI
    " toolbar icons send instead of BACK/EXIT/CANC - catching both.
    WHEN 'BACK' OR 'EXIT' OR 'CANC' OR 'CANCEL' OR '&F03' OR '&F12' OR '&F15'.
      go_cc_selparm->free( ).
      CLEAR go_cc_selparm.
      CLEAR go_alv_selparm.
      LEAVE TO SCREEN 0.
  ENDCASE.
ENDMODULE.
