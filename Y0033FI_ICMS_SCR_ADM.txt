* Applied once here too, not just on MOD_CHG (see SET_STATUS_FILTER_DEFAULTS
* below) - the individual filter checkboxes all carry PARAMETERS-level
* DEFAULT 'X', so without this the very first screen display showed
* every one of them checked regardless of which mode (P_MADM by
* default) is actually selected, until the user touched a radio button
* and triggered MOD_CHG for the first time.
  PERFORM set_status_filter_defaults.

AT SELECTION-SCREEN OUTPUT.
  LOOP AT SCREEN.
    IF screen-group1 = 'FS' AND p_mnew <> 'X'.
      screen-active = 0.  " hide entirely - only applies when searching new candidates
      MODIFY SCREEN.
    ENDIF.
    IF screen-group1 = 'FA' AND p_mnew <> 'X' AND p_auto <> 'X'.
      screen-active = 0.  " search-only filters/model switches - also needed by Auto Approved (P_AUTO)
      MODIFY SCREEN.
    ENDIF.
    IF screen-group1 = 'FV' AND p_mview <> 'X' AND p_madm <> 'X'.
      screen-active = 0.  " hide entirely - only applies to P_MVIEW/P_MADM
      MODIFY SCREEN.
    ENDIF.
    IF screen-group1 = 'MV' AND p_mview <> 'X'.
      screen-active = 0.  " hide the "View" master checkboxes outside P_MVIEW
      MODIFY SCREEN.
    ENDIF.
    IF screen-group1 = 'MA' AND p_madm <> 'X'.
      screen-active = 0.  " hide the "Admin" master checkboxes outside P_MADM
      MODIFY SCREEN.
    ENDIF.
  ENDLOOP.

AT SELECTION-SCREEN.
* ONLI = the user pressed Execute (F8), not just clicked a radio
* button/checkbox - only THEN is P_BUKRS actually required (and its
* authorization worth checking). Background jobs don't go through AT
* SELECTION-SCREEN interactively, so this only affects foreground
* execution - see the unconditional check in START-OF-SELECTION for
* the background-safe equivalent of both.
  IF sy-ucomm = 'ONLI'.
    IF p_bukrs IS INITIAL.
      MESSAGE 'Enter a Company Code' TYPE 'E'.
    ENDIF.
    PERFORM check_bukrs_authority USING p_bukrs.
    IF p_mnew = 'X' OR p_auto = 'X'.
      PERFORM check_search_filters.
    ENDIF.
  ENDIF.

* Mode just switched (MOD_CHG fires for any change within the P_MNEW/
* P_MVIEW/P_MADM/P_MPARM/P_MLANE radio group, not only when P_MNEW
* itself is clicked) - reset each mode's own status filter to its
* defined default, exactly as if the screen were opened fresh in that
* mode. This intentionally overwrites any manual per-checkbox tweak
* made before switching away. P_MPARM/P_MLANE have no status filter of
* their own, so nothing to reset there.
  IF sy-ucomm = 'MOD_CHG'.
    PERFORM set_status_filter_defaults.
  ENDIF.

* "Master" checkboxes (B3B) - group-toggle the real filter checkboxes
* when clicked. See the block comment above P_MVG1 for what each
* group covers.
  CASE sy-ucomm.
    WHEN 'MVG1'.
      p_vca = p_mvg1.  p_vcd = p_mvg1.
    WHEN 'MVG3'.
      p_vpo = p_mvg3.
    WHEN 'MVG2'.
      p_vrj = p_mvg2.  p_vap = p_mvg2.  p_vpf = p_mvg2.  p_vcb = p_mvg2.  p_vpg = p_mvg2.
      p_vrr = p_mvg2.  p_vrv = p_mvg2.  p_ver = p_mvg2.
    WHEN 'MAG1'.
      p_vap = p_mag1.  p_vpf = p_mag1.  p_vrr = p_mag1.
    WHEN 'MAG2'.
      p_vca = p_mag2.  p_vrj = p_mag2.  p_vpg = p_mag2.  p_vcd = p_mag2.
      p_vpo = p_mag2.  p_vrv = p_mag2.  p_ver = p_mag2.  p_vcb = p_mag2.
  ENDCASE.

* Keep all five master checkboxes honest: recompute each from its
* actual children on EVERY round trip (not just when the master
* itself was clicked) - otherwise switching between Search/View/Admin
* (or any other trip to the server, e.g. pressing Execute) kept
* showing whichever value the master had last, even after a child was
* unchecked individually, instead of tracking its children's real
* combined state.
  p_mvg1 = COND #( WHEN p_vca = abap_true AND p_vcd = abap_true
                    THEN abap_true ELSE abap_false ).
  p_mvg3 = COND #( WHEN p_vpo = abap_true THEN abap_true ELSE abap_false ).
  p_mvg2 = COND #( WHEN p_vrj = abap_true AND p_vap = abap_true AND p_vpf = abap_true AND p_vcb = abap_true
                         AND p_vpg = abap_true AND p_vrr = abap_true AND p_vrv = abap_true
                         AND p_ver = abap_true
                    THEN abap_true ELSE abap_false ).
  p_mag1 = COND #( WHEN p_vap = abap_true AND p_vpf = abap_true AND p_vrr = abap_true
                    THEN abap_true ELSE abap_false ).
  p_mag2 = COND #( WHEN p_vca = abap_true AND p_vrj = abap_true AND p_vpg = abap_true
                         AND p_vcd = abap_true AND p_vpo = abap_true AND p_vrv = abap_true
                         AND p_ver = abap_true AND p_vcb = abap_true
                    THEN abap_true ELSE abap_false ).

*&---------------------------------------------------------------------*
*&      Form  SET_STATUS_FILTER_DEFAULTS
*&---------------------------------------------------------------------*
* Resets the per-status filter checkboxes (B3B) to the current mode's
* default set. Called both from INITIALIZATION (so the very first
* screen display already matches the default mode, P_MADM, instead of
* showing every checkbox checked - their PARAMETERS-level DEFAULT 'X'
* only makes sense as a safe fallback) and from AT SELECTION-SCREEN on
* MOD_CHG (every later mode switch). P_MPARM/P_MLANE have no status
* filter of their own, so nothing to reset for them.
*---------------------------------------------------------------------*
FORM set_status_filter_defaults.
  IF p_mnew = 'X'.
    p_sre = abap_true.  p_snc = abap_false.  p_ser = abap_false.
  ELSEIF p_mview = 'X'.
    p_vca = abap_true.   p_vcd = abap_true.                    " needs my action (010, 050)
    p_vpo = abap_false.                                        " reversal (040) - off by default
    p_vrj = abap_false.  p_vap = abap_false.  p_vpf = abap_false.  p_vcb = abap_false.
    p_vpg = abap_false.  p_vrr = abap_false.  p_vrv = abap_false.  p_ver = abap_false. " others
  ELSEIF p_madm = 'X'.
    p_vap = abap_true.   p_vpf = abap_true.   p_vrr = abap_true.  " needs admin action (020/021, 025, 060)
    p_vca = abap_false.  p_vrj = abap_false.  p_vpg = abap_false.
    p_vcd = abap_false.  p_vpo = abap_false.  p_vrv = abap_false.  p_ver = abap_false.
    p_vcb = abap_false.  " others (045 included - no admin action exists for Cancelled Before Posting)
  ENDIF.
ENDFORM.
