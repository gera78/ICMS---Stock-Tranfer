*&---------------------------------------------------------------------*
*&      Form  AUTO_APPROVE_CANDIDATES
*&---------------------------------------------------------------------*
* Admin run with P_AUTO (Auto Approved) checked. Runs the same search
* chain as "Search New Candidates" (P_MNEW, incl. the P_MODE1/P_MODE2
* model switches) and persists every row that comes out as
* READY_TO_SAVE (900) directly as AUTO_APPROVED (021) - i.e. the
* manual Save (010) + Approve (020) steps done in one go, with no user
* selection. Same safety checks as SAVE_CANDIDATES (lock + active-
* cycle re-check + next SEQNR) and one log row per item.
* POST_RELEASED_ITEMS then picks AUTO_APPROVED up exactly like
* APPROVED.
*
* Same guard as CHECK_SEARCH_FILTERS (at least one filter besides
* P_BUKRS), but never raised as TYPE E here - in a background job
* that would abort the rest of the cycle (post/reversal), so the
* auto-approval step is just skipped with a message instead.
*
* GIT_ITEM is used as the work table for the search chain and is
* cleared at the end, so the mode selected on screen starts clean.
*---------------------------------------------------------------------*
FORM auto_approve_candidates USING pu_bukrs   TYPE bukrs
                                   pu_id_exec TYPE char20.

  DATA lv_filters_ok TYPE abap_bool.
  PERFORM has_search_filters CHANGING lv_filters_ok.
  IF lv_filters_ok = abap_false.
    IF sy-batch = abap_true.
      WRITE: / 'Auto Approved skipped: fill in at least one filter besides Company Code'.
    ELSE.
      MESSAGE 'Auto Approved skipped: fill in at least one filter besides Company Code' TYPE 'S' DISPLAY LIKE 'W'.
    ENDIF.
    RETURN.
  ENDIF.

  CLEAR git_item.
  IF p_mode1 = abap_true.
    PERFORM select_candidate_documents USING pu_bukrs.
  ENDIF.
  IF p_mode2 = abap_true.
    PERFORM select_transfer_candidates USING pu_bukrs.
  ENDIF.
  PERFORM read_items_and_icms.
  PERFORM determine_benefit_percentage.
  PERFORM filter_active_cycle.
  PERFORM calculate_credit.
  PERFORM flag_not_candidates.

  DATA lt_ready TYPE ty_it_item.
  LOOP AT git_item INTO DATA(ls_cand) WHERE status = gc_status-ready_to_save.
    APPEND ls_cand TO lt_ready.
  ENDLOOP.
  CLEAR git_item.

  DATA lv_saved TYPE i.
  DATA lv_skip  TYPE i.

  LOOP AT lt_ready INTO DATA(ls_sel).
    DATA lv_subrc TYPE sy-subrc.
    PERFORM enqueue_item USING ls_sel-docnum ls_sel-itmnum 0 CHANGING lv_subrc.
    IF lv_subrc <> 0.
      lv_skip = lv_skip + 1.
      CONTINUE.
    ENDIF.

    " Same race protection (and same "closed" statuses) as
    " SAVE_CANDIDATES/FILTER_ACTIVE_CYCLE.
    SELECT COUNT(*) FROM y0033fi_icm_item
      INTO @DATA(lv_active_count)
      WHERE docnum = @ls_sel-docnum
        AND itmnum = @ls_sel-itmnum
        AND status <> @gc_status-reversed
        AND status <> @gc_status-rejected
        AND status <> @gc_status-cancelled_pre_post.
    IF lv_active_count > 0.
      lv_skip = lv_skip + 1.
      PERFORM dequeue_item USING ls_sel-docnum ls_sel-itmnum 0.
      CONTINUE.
    ENDIF.

    DATA lv_seqnr TYPE n LENGTH 3.
    PERFORM next_seqnr USING ls_sel-docnum ls_sel-itmnum CHANGING lv_seqnr.
    ls_sel-seqnr    = lv_seqnr.
    ls_sel-status   = gc_status-auto_approved.
    ls_sel-cre_user = sy-uname.
    ls_sel-cre_date = sy-datum.
    ls_sel-cre_time = sy-uzeit.
    ls_sel-apr_user = sy-uname.
    ls_sel-apr_date = sy-datum.
    ls_sel-apr_time = sy-uzeit.
    ls_sel-id_exec  = pu_id_exec.

    INSERT y0033fi_icm_item FROM @( CORRESPONDING y0033fi_icm_item( ls_sel ) ).
    IF sy-subrc = 0.
      PERFORM write_log USING ls_sel gc_action-auto_approved '000' gc_status-auto_approved
                               'Candidate auto-approved by admin run'.
      IF ls_sel-edoc_guid IS NOT INITIAL.
        UPDATE y0033fi_icm_edoc SET seqnr = ls_sel-seqnr
          WHERE docnum = ls_sel-docnum AND edoc_guid = ls_sel-edoc_guid.
      ENDIF.
      lv_saved = lv_saved + 1.
    ELSE.
      lv_skip = lv_skip + 1.
    ENDIF.

    PERFORM dequeue_item USING ls_sel-docnum ls_sel-itmnum lv_seqnr.
  ENDLOOP.

  COMMIT WORK AND WAIT.

  IF gv_api_mode = abap_true.
    RETURN. " API: results go back through RETURN tables, no list/message output
  ELSEIF sy-batch = abap_true.
    WRITE: / 'Auto Approved:', lv_saved, 'item(s) - skipped:', lv_skip.
  ELSE.
    MESSAGE s010(y0033icms) WITH lv_saved lv_skip.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  POST_RELEASED_ITEMS
*&---------------------------------------------------------------------*
* One accounting document per nota fiscal (DOCNUM), as agreed - not
* aggregated across notas. Within a document, one debit/credit pair
* PER ITEM (not per nota total), so BUZEI_FI always points to the
* exact line for that item - full traceability, at the cost of more
* lines. If a single nota fiscal has more items than the 999-line
* limit allows (max 499 debit/credit pairs), it is automatically
* split into several documents for that same nota fiscal.
*
* TODO (Q13): document type, currency handling detail still to
* confirm with FI beyond what is fixed here (doc type from constant
* GC_DOC_TYPE, currency from T001 via GV_WAERS).
* TODO (Q16): accounting field for the nota fiscal reference - uses
* ASSIGNMENT (ZUONR) as DOCNUM+ITMNUM below; confirm with FI.
*---------------------------------------------------------------------*
FORM post_released_items.

  " AUTO_APPROVED (021, see AUTO_APPROVE_CANDIDATES) is posted exactly
  " like a manually APPROVED (020) item.
  SELECT * FROM y0033fi_icm_item
    INTO TABLE @DATA(lt_approved)
    WHERE status = @gc_status-approved
       OR status = @gc_status-auto_approved
       OR status = @gc_status-post_failed.

  " Online (screen 200 "Create Document(s)" button): only post what the
  " admin actually marked with SEL in the ALV - GIT_ITEM holds exactly
  " this run's admin review data, so any row not selected there (or no
  " longer present) is dropped before posting. The background job has
  " no GIT_ITEM/screen at all and must keep posting every matching item
  " regardless of any screen state, so this only applies interactively.
  IF sy-batch = abap_false.
    LOOP AT lt_approved INTO DATA(ls_check).
      READ TABLE git_item WITH KEY docnum = ls_check-docnum
                                    itmnum = ls_check-itmnum
                                    seqnr  = ls_check-seqnr
        INTO DATA(ls_git_check).
      IF sy-subrc <> 0 OR ls_git_check-sel <> abap_true.
        DELETE lt_approved.
      ENDIF.
    ENDLOOP.
  ENDIF.

  " Cancelled-nota check + one document per nota fiscal - shared with
  " the single-DOCNUM API (API_AUTO_POST, Y0033FI_ICMS_F03), see
  " POST_ITEM_LIST below. Items skipped as CANCELLED_PRE_POST are
  " removed from LT_APPROVED, so the checks below only see what was
  " actually sent to posting.
  PERFORM post_item_list CHANGING lt_approved.

  IF lt_approved IS INITIAL.
    " Nothing to post is a normal, expected outcome (not every run has
    " approved items, or - online - nothing was marked) - TYPE E would
    " flag the background job as ended in error in SM37 even though
    " nothing actually went wrong, so it is only used interactively.
    IF sy-batch = abap_true.
      MESSAGE s007(y0033icms).
    ELSE.
      MESSAGE e007(y0033icms).
    ENDIF.
    RETURN.
  ENDIF.

  " Summary message - POST_ONE_DOCUMENT never reports back to this FORM
  " (it updates Y0033FI_ICM_ITEM directly, chunk by chunk), so the only
  " reliable way to know the outcome is to re-read the same items this
  " run started with and see where each one landed.
  DATA lv_posted_ok   TYPE i.
  DATA lv_posted_fail TYPE i.
  LOOP AT lt_approved INTO DATA(ls_result).
    SELECT SINGLE status FROM y0033fi_icm_item
      INTO @DATA(lv_result_status)
      WHERE docnum = @ls_result-docnum AND itmnum = @ls_result-itmnum AND seqnr = @ls_result-seqnr.
    IF lv_result_status = gc_status-posted.
      lv_posted_ok = lv_posted_ok + 1.
    ELSE.
      lv_posted_fail = lv_posted_fail + 1.
    ENDIF.
  ENDLOOP.

  IF lv_posted_fail > 0.
    MESSAGE s022(y0033icms) WITH lv_posted_ok lv_posted_fail DISPLAY LIKE 'E'.
  ELSE.
    MESSAGE s022(y0033icms) WITH lv_posted_ok lv_posted_fail.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  POST_ITEM_LIST
*&---------------------------------------------------------------------*
* Posts a list of APPROVED/AUTO_APPROVED/POST_FAILED items: skips (and
* flags as CANCELLED_PRE_POST) those whose nota fiscal is already
* cancelled, then posts one accounting document per nota fiscal, split
* in chunks of up to LC_MAX_PAIRS items. Extracted unchanged from
* POST_RELEASED_ITEMS so the single-DOCNUM API (Y0033FI_ICMS_F03) posts
* exactly the same way. PC_ITEMS comes back without the cancelled ones.
*---------------------------------------------------------------------*
FORM post_item_list CHANGING pc_items TYPE y0033fi_icm_item_tt.

  CONSTANTS lc_max_pairs TYPE i VALUE 499.  " 999 / 2 lines per document

  " Skip items whose nota fiscal was already cancelled (J_1BNFDOC-CANCEL)
  " by the time this run tries to post them - posting first and relying
  " on CHECK_NF_CANCELLATION/reversal to clean up afterward would create
  " a real FI document for a nota that's already void. Flagged with its
  " own status (CANCELLED_PRE_POST) instead, so these never reach GL and
  " the user can filter "victims of cancellation" before posting.
  IF pc_items IS NOT INITIAL.
    " Structured single-field driving table (not a bare elementary
    " itab) for FOR ALL ENTRIES - same workaround already used for
    " LT_NFE_KEYS in SELECT_CANDIDATE_DOCUMENTS (Y0033FI_ICMS_F01).
    TYPES: BEGIN OF ty_docnum_key,
             docnum TYPE j_1bnfdoc-docnum,
           END OF ty_docnum_key.
    DATA lt_docnum_check TYPE STANDARD TABLE OF ty_docnum_key WITH DEFAULT KEY.
    LOOP AT pc_items INTO DATA(ls_appr_chk).
      APPEND VALUE #( docnum = ls_appr_chk-docnum ) TO lt_docnum_check.
    ENDLOOP.
    SORT lt_docnum_check BY docnum.
    DELETE ADJACENT DUPLICATES FROM lt_docnum_check COMPARING docnum.

    DATA lt_cancelled_docnum TYPE STANDARD TABLE OF j_1bnfdoc-docnum WITH DEFAULT KEY.
    SELECT docnum FROM j_1bnfdoc
      INTO TABLE @lt_cancelled_docnum
      FOR ALL ENTRIES IN @lt_docnum_check
      WHERE docnum = @lt_docnum_check-docnum
        AND cancel = @abap_true.
    SORT lt_cancelled_docnum BY table_line.

    DATA lv_victims TYPE i.
    LOOP AT pc_items INTO DATA(ls_victim).
      READ TABLE lt_cancelled_docnum WITH KEY table_line = ls_victim-docnum
        TRANSPORTING NO FIELDS BINARY SEARCH.
      CHECK sy-subrc = 0.

      DATA lv_subrc_v TYPE sy-subrc.
      PERFORM enqueue_item USING ls_victim-docnum ls_victim-itmnum ls_victim-seqnr CHANGING lv_subrc_v.
      IF lv_subrc_v <> 0.
        CONTINUE.
      ENDIF.

      " STATUS = LS_VICTIM-STATUS guards against a stale batch, same as
      " REQUEST_REVERSAL - if another session already moved this item on
      " since LT_APPROVED was read, nothing is changed/logged here.
      UPDATE y0033fi_icm_item SET status = @gc_status-cancelled_pre_post
        WHERE docnum = @ls_victim-docnum AND itmnum = @ls_victim-itmnum AND seqnr = @ls_victim-seqnr
          AND status = @ls_victim-status.

      IF sy-subrc = 0.
        PERFORM write_log USING ls_victim gc_action-cancelled_pre_post ls_victim-status gc_status-cancelled_pre_post
                                 'Nota fiscal already cancelled - skipped before posting'.
        lv_victims = lv_victims + 1.
      ENDIF.

      PERFORM dequeue_item USING ls_victim-docnum ls_victim-itmnum ls_victim-seqnr.
      DELETE pc_items.
    ENDLOOP.

    IF lv_victims > 0.
      COMMIT WORK AND WAIT.
    ENDIF.
  ENDIF.

  CHECK pc_items IS NOT INITIAL.

  SORT pc_items BY docnum itmnum.

  " Distinct list of DOCNUMs to process. Deliberately classic/simple -
  " no LOOP...GROUP BY/REFERENCE INTO/LOOP AT GROUP - a real posting
  " run showed one item duplicated into the accounting document and
  " another dropped entirely (never posted, left stuck in POSTING).
  " That grouping mechanism was not conclusively proven to be the
  " cause, but it is newer and less battle-tested than a plain
  " READ TABLE/LOOP...WHERE, so it is not worth the risk here.
  DATA lt_docnums TYPE STANDARD TABLE OF j_1bdocnum WITH DEFAULT KEY.
  LOOP AT pc_items INTO DATA(ls_appr).
    READ TABLE lt_docnums WITH KEY table_line = ls_appr-docnum TRANSPORTING NO FIELDS.
    IF sy-subrc <> 0.
      APPEND ls_appr-docnum TO lt_docnums.
    ENDIF.
  ENDLOOP.

  LOOP AT lt_docnums INTO DATA(lv_docnum).

    DATA lt_group TYPE y0033fi_icm_item_tt.
    CLEAR lt_group.
    LOOP AT pc_items INTO DATA(ls_member) WHERE docnum = lv_docnum.
      APPEND ls_member TO lt_group.
    ENDLOOP.

    " Split into chunks of LC_MAX_PAIRS items each (same nota fiscal)
    DATA lv_from TYPE i.
    DATA lv_to   TYPE i.
    DATA lv_idx  TYPE i.
    DATA lt_chunk TYPE y0033fi_icm_item_tt.
    DATA ls_chunk_line TYPE y0033fi_icm_item.
    " Explicit reset every time this DOCNUM starts - a DATA... VALUE
    " clause only initializes ONCE for the whole form, NOT on every
    " pass through this outer LOOP.
    lv_from = 1.
    WHILE lv_from <= lines( lt_group ).
      lv_to = lv_from + lc_max_pairs - 1.
      IF lv_to > lines( lt_group ).
        lv_to = lines( lt_group ).
      ENDIF.
      CLEAR lt_chunk.
      lv_idx = lv_from.
      WHILE lv_idx <= lv_to.
        CLEAR ls_chunk_line.
        READ TABLE lt_group INTO ls_chunk_line INDEX lv_idx.
        APPEND ls_chunk_line TO lt_chunk.
        lv_idx = lv_idx + 1.
      ENDWHILE.
      PERFORM post_one_document USING lt_chunk.
      lv_from = lv_to + 1.
    ENDWHILE.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_POSTING_LINES
*&---------------------------------------------------------------------*
* One debit/credit pair per item (GL_DEBIT/GL_CREDIT from
* Y0033FI_ICM_PARM, amount = CREDIT_AMOUNT). Extracted unchanged from
* POST_ONE_DOCUMENT so the lines handed to another process by
* GET_POSTING_DATA (Y0033FI_ICMS_F03) are exactly the ones this
* program would post itself. PU_ITEMNO_OFFSET shifts ITEMNO_ACC so the
* lines can be appended after the caller's own lines (0 = start at 1).
* PC_BUZEI maps each item to its debit line's ITEMNO_ACC.
*---------------------------------------------------------------------*
FORM build_posting_lines USING    pu_chunk         TYPE y0033fi_icm_item_tt
                                  pu_itemno_offset TYPE posnr_acc
                         CHANGING pc_gl            TYPE ty_it_acc_gl
                                  pc_cur           TYPE ty_it_acc_cur
                                  pc_buzei         TYPE ty_it_buzei_map.

  DATA: ls_gl     TYPE bapiacgl09,
        ls_cur    TYPE bapiaccr09,
        ls_buzei  TYPE ty_buzei_map,
        lv_itemno TYPE posnr_acc.

  CLEAR: pc_gl, pc_cur, pc_buzei.
  lv_itemno = pu_itemno_offset.

  LOOP AT pu_chunk INTO DATA(ls_it).
    lv_itemno = lv_itemno + 1.

    DATA lv_ref_docnum TYPE j_1bdocnum.
    PERFORM resolve_ref_docnum USING ls_it CHANGING lv_ref_docnum.
    DATA lv_ref_werks TYPE werks_d.
    PERFORM resolve_ref_werks USING ls_it CHANGING lv_ref_werks.

    CLEAR ls_gl.
    ls_gl-itemno_acc = lv_itemno.
    ls_gl-gl_account = gv_gl_debit.
    ls_gl-comp_code  = ls_it-bukrs.
    " Confirmed via a live field dump of this system's BAPIACGL09:
    " the business place field here is called BUSINESSPLACE, not
    " BUPLA (older SAP releases/discussions don't have this field
    " at all on BAPIACGL09 - this release does).
    ls_gl-businessplace = ls_it-branch.
    " TAX_CODE removed - confirmed by a real test that this field
    " makes the posting fail for this account (blank or not).
    ls_gl-profit_ctr = ls_it-prctr.   " sourced from J_1BNFLIN-PRCTR - may be blank
    ls_gl-material   = ls_it-matnr.
    ls_gl-plant      = lv_ref_werks.  " TODO (Q16) - see RESOLVE_REF_WERKS
    ls_gl-alloc_nmbr = |{ lv_ref_docnum }{ ls_it-itmnum }|.  " TODO (Q16) - see RESOLVE_REF_DOCNUM
    ls_gl-item_text  = |ICMS benefit { lv_ref_docnum }/{ ls_it-itmnum }|.
    APPEND ls_gl TO pc_gl.

    CLEAR ls_cur.
    ls_cur-itemno_acc = lv_itemno.
    ls_cur-currency   = ls_it-waers.
    ls_cur-amt_doccur = ls_it-credit_amount.
    APPEND ls_cur TO pc_cur.

    CLEAR ls_buzei.
    ls_buzei-docnum = ls_it-docnum.
    ls_buzei-itmnum = ls_it-itmnum.
    ls_buzei-seqnr  = ls_it-seqnr.
    ls_buzei-buzei  = lv_itemno.
    APPEND ls_buzei TO pc_buzei.

    lv_itemno = lv_itemno + 1.
    CLEAR ls_gl.
    ls_gl-itemno_acc = lv_itemno.
    ls_gl-gl_account = gv_gl_credit.
    ls_gl-comp_code  = ls_it-bukrs.
    ls_gl-businessplace = ls_it-branch.
    ls_gl-profit_ctr = ls_it-prctr.
    ls_gl-material   = ls_it-matnr.
    ls_gl-plant      = lv_ref_werks.  " TODO (Q16) - see RESOLVE_REF_WERKS
    ls_gl-alloc_nmbr = |{ lv_ref_docnum }{ ls_it-itmnum }|.  " TODO (Q16) - see RESOLVE_REF_DOCNUM
    ls_gl-item_text  = |ICMS benefit { lv_ref_docnum }/{ ls_it-itmnum }|.
    APPEND ls_gl TO pc_gl.

    CLEAR ls_cur.
    ls_cur-itemno_acc = lv_itemno.
    ls_cur-currency   = ls_it-waers.
    ls_cur-amt_doccur = ls_it-credit_amount * -1.
    APPEND ls_cur TO pc_cur.
  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  RESOLVE_REF_DOCNUM
*&---------------------------------------------------------------------*
* Returns the DOCNUM to use on the accounting reference fields
* (ALLOC_NMBR/ITEM_TEXT, see POST_ONE_DOCUMENT) for one item -
* PU_ITEM-DOCNUM (today's behavior) unless GV_USE_ORIG_DOCNUM_REF
* (Y0033FI_ICMS_TOP, TODO Q16) is set, in which case the ORIGIN nota
* fiscal's own J_1BNFDOC row (DIRECT = 2, same NFENUM+SERIES+BUKRS) is
* looked up instead. Falls back to PU_ITEM-DOCNUM whenever that lookup
* finds nothing, so this switch can never block a posting that would
* otherwise succeed. No DDIC change - resolved here, at posting time
* only, never persisted on Y0033FI_ICM_ITEM.
*---------------------------------------------------------------------*
FORM resolve_ref_docnum USING pu_item TYPE y0033fi_icm_item
                     CHANGING pc_docnum TYPE j_1bdocnum.

  pc_docnum = pu_item-docnum.

  CHECK gv_use_orig_docnum_ref = abap_true.

  SELECT SINGLE docnum FROM j_1bnfdoc
    INTO @DATA(lv_orig_docnum)
    WHERE bukrs  = @pu_item-bukrs
      AND nfenum = @pu_item-nfenum
      AND series = @pu_item-series
      AND direct = '2'.
  IF sy-subrc = 0.
    pc_docnum = lv_orig_docnum.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  RESOLVE_REF_WERKS
*&---------------------------------------------------------------------*
* Same concept as RESOLVE_REF_DOCNUM above, for the accounting
* document's plant (BAPIACGL09-PLANT, see POST_ONE_DOCUMENT):
* PU_ITEM-WERKS_DEST (today's behavior, the destination plant) unless
* GV_USE_ORIG_WERKS_REF (Y0033FI_ICMS_TOP) is set, in which case the
* ORIGIN nota fiscal's own branch (J_1BNFDOC DIRECT = 2, same
* NFENUM+SERIES+BUKRS) is resolved to its plant via T001W-J_1BBRANCH -
* the same branch->plant lookup SELECT_TRANSFER_CANDIDATES already
* does for the destination side (Y0033FI_ICMS_F01). Falls back to
* PU_ITEM-WERKS_DEST whenever either lookup finds nothing, so this
* switch can never block a posting that would otherwise succeed.
*---------------------------------------------------------------------*
FORM resolve_ref_werks USING pu_item TYPE y0033fi_icm_item
                     CHANGING pc_werks TYPE werks_d.

  pc_werks = pu_item-werks_dest.

  CHECK gv_use_orig_werks_ref = abap_true.

  SELECT SINGLE branch FROM j_1bnfdoc
    INTO @DATA(lv_orig_branch)
    WHERE bukrs  = @pu_item-bukrs
      AND nfenum = @pu_item-nfenum
      AND series = @pu_item-series
      AND direct = '2'.
  CHECK sy-subrc = 0.

  SELECT SINGLE werks FROM t001w
    INTO @DATA(lv_orig_werks)
    WHERE j_1bbranch = @lv_orig_branch.
  IF sy-subrc = 0.
    pc_werks = lv_orig_werks.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  POST_ONE_DOCUMENT
*&---------------------------------------------------------------------*
* Posts a single accounting document for a chunk of items belonging
* to the SAME nota fiscal (DOCNUM). Handles the PSTG transitional
* lock state and the period-closed fallback (BUDAT -> today).
*
* TAX_CODE and PROFIT_CTR are set from MWSKZ/PRCTR (both confirmed
* present in J_1BNFLIN, though often blank in practice - confirmed via
* a live field dump). BUSINESSPLACE (business place) is set from
* BRANCH - confirmed present on BAPIACGL09 in this system via the same
* field dump (older SAP discussions call this field BUPLA and say it
* doesn't exist on this structure at all - this release has it under
* a different name).
*---------------------------------------------------------------------*
FORM post_one_document USING pu_chunk TYPE y0033fi_icm_item_tt.

  DATA: ls_header  TYPE bapiache09,
        lt_gl      TYPE ty_it_acc_gl,
        lt_cur     TYPE ty_it_acc_cur,
        lt_return  TYPE TABLE OF bapiret2,
        lv_locked  TYPE abap_bool VALUE abap_true,
        ls_ok      TYPE bapiret2,
        lv_belnr   TYPE belnr_d,
        lv_gjahr   TYPE gjahr,
        ls_err     TYPE bapiret2.

  " BUZEI lookup, keyed by DOCNUM+ITMNUM+SEQNR - built via APPEND only
  " (never MODIFY-in-place on PU_CHUNK mid-loop, which a real test
  " showed producing duplicated GL lines). Read back further down
  " when marking items POSTED, instead of relying on PU_CHUNK itself
  " carrying the value forward.
  DATA ls_buzei TYPE ty_buzei_map.
  DATA lt_buzei TYPE ty_it_buzei_map.

  " Lock and mark every item in the chunk as PSTG before posting, so a
  " parallel run cannot pick the same items up.
  LOOP AT pu_chunk INTO DATA(ls_it).
    DATA lv_subrc TYPE sy-subrc.
    PERFORM enqueue_item USING ls_it-docnum ls_it-itmnum ls_it-seqnr CHANGING lv_subrc.
    IF lv_subrc <> 0.
      lv_locked = abap_false.
      EXIT.
    ENDIF.
    UPDATE y0033fi_icm_item SET status = @gc_status-posting
      WHERE docnum = @ls_it-docnum AND itmnum = @ls_it-itmnum AND seqnr = @ls_it-seqnr.
  ENDLOOP.

  IF lv_locked = abap_false.
    " Could not lock every item - release what was taken and skip
    " this chunk this run; it stays APR and will be retried later.
    LOOP AT pu_chunk INTO ls_it.
      PERFORM dequeue_item USING ls_it-docnum ls_it-itmnum ls_it-seqnr.
    ENDLOOP.
    ROLLBACK WORK.
    RETURN.
  ENDIF.
  COMMIT WORK AND WAIT. " persist PSTG before the BAPI call

  READ TABLE pu_chunk INTO DATA(ls_any) INDEX 1.

* If any item in this chunk is already POST_FAILED, this is a retry -
* we already know posting with the nota fiscal's own date failed
* before (see GC_STATUS-POST_FAILED), so there is no point trying
* that same date again. Go straight to today's date (current period)
* instead of wasting a cycle on a date we already know will fail.
* BLDAT (document date) never changes - only BUDAT (posting date), same
* as the existing period-closed fallback below.
  DATA(lv_bldat) = ls_any-docdat.
  DATA(lv_is_retry) = xsdbool( line_exists( pu_chunk[ status = gc_status-post_failed ] ) ).
  DATA(lv_budat) = COND #( WHEN lv_is_retry = abap_true THEN sy-datum ELSE ls_any-docdat ).
  DATA(lv_fallback) = lv_is_retry.

  " BKPF-TCODE otherwise records SE38 (this report's own execution
  " context, not a real transaction) - the previous attempt to fix
  " this via EXTENSION1/FIELD1='BKPF-TCODE' broke the posting itself
  " ("Inconsistent FI/CO document header data"), since EXTENSION1
  " needs a matching FI BAdI to interpret it, which doesn't exist
  " here. SY-TCODE itself is only read (not enforced) by the posting
  " kernel, so swapping it just for this BAPI call and restoring it
  " right after is safe and doesn't touch document consistency.
  DATA(lv_saved_tcode) = sy-tcode.
  IF gv_program_id IS NOT INITIAL.
    sy-tcode = gv_program_id.
  ENDIF.

  DO 2 TIMES.
    CLEAR: ls_header, lt_gl, lt_cur, lt_return, ls_err, lt_buzei.

    ls_header-comp_code  = ls_any-bukrs.
    ls_header-doc_date   = lv_bldat.
    ls_header-pstng_date = lv_budat.
    ls_header-fisc_year  = lv_budat(4).
    ls_header-doc_type   = gv_doc_type.
    ls_header-username   = sy-uname.
    ls_header-header_txt = |ICMS transfer { ls_any-docnum }|.
    " XBLNR = nota fiscal number + series, e.g. "000273398-1" -
    " matches the format confirmed in a real posting (BKPF-XBLNR),
    " not our internal DOCNUM (J_1BDOCNUM).
    ls_header-ref_doc_no = |{ ls_any-nfenum }-{ ls_any-series }|.

    " GL/currency lines - built by BUILD_POSTING_LINES, shared with the
    " API that hands these same lines to another process instead of
    " posting them (GET_POSTING_DATA, Y0033FI_ICMS_F03).
    PERFORM build_posting_lines USING pu_chunk 0
                                CHANGING lt_gl lt_cur lt_buzei.

    CALL FUNCTION 'BAPI_ACC_DOCUMENT_POST'
      EXPORTING
        documentheader = ls_header
      TABLES
        accountgl      = lt_gl
        currencyamount = lt_cur
        return         = lt_return.

    READ TABLE lt_return INTO ls_err WITH KEY type = 'E'.
    IF sy-subrc <> 0.
      READ TABLE lt_return INTO ls_err WITH KEY type = 'A'.
    ENDIF.
    IF sy-subrc <> 0.
      EXIT. " posted with no error - leave the retry loop
    ENDIF.

    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

    " Period-closed heuristic: message class F5, number 053 is the
    " standard "posting period not open" error. Retry immediately (same
    " run, not the next background cycle) with BUDAT = today, which is
    " virtually always in an open period.
    "
    " Search the WHOLE LT_RETURN table for F5/053, not just LS_ERR (the
    " primary error message used above for the POST_FAILED log/MSG) -
    " BAPI_ACC_DOCUMENT_POST commonly wraps the real cause behind a
    " generic "Error in document: BKPFF ..." message that also comes
    " back as TYPE E, and when THAT one lands first in the table (very
    " common), LS_ERR held it instead of the real F5/053 entry, so this
    " check silently never matched and the retry never happened.
    IF lv_fallback = abap_false AND line_exists( lt_return[ id = 'F5' number = '053' ] ).
      lv_budat    = sy-datum.
      lv_fallback = abap_true.
      CONTINUE.
    ELSE.
      EXIT.
    ENDIF.
  ENDDO.

  sy-tcode = lv_saved_tcode. " restore immediately - never leak the fake value beyond this BAPI call

  IF ls_err IS NOT INITIAL.
    PERFORM smart_case CHANGING ls_err-message.
    " Posting failed - release the chunk to POST_FAILED (not back to
    " APPROVED - that would look identical to "never attempted"; see
    " design discussion on GC_STATUS-POST_FAILED) with the primary
    " error message shown on the item, so it can be corrected and
    " retried - POST_RELEASED_ITEMS picks up POST_FAILED items the
    " same as APPROVED ones, whether run interactively or in
    " background. Exactly ONE log row per item records the real status
    " transition (POSTING -> POST_FAILED); every OTHER message the
    " BAPI returned is logged as BAPI_DETAIL below, with STATUS_FROM =
    " STATUS_TO, since it's supplementary detail about the SAME
    " failure, not a separate transition.
    LOOP AT pu_chunk INTO ls_it.
      UPDATE y0033fi_icm_item SET status = @gc_status-post_failed,
                                  msg    = @ls_err-message
        WHERE docnum = @ls_it-docnum AND itmnum = @ls_it-itmnum AND seqnr = @ls_it-seqnr.
      PERFORM write_log USING ls_it gc_action-error gc_status-posting gc_status-post_failed
                               ls_err-message.
      PERFORM dequeue_item USING ls_it-docnum ls_it-itmnum ls_it-seqnr.
    ENDLOOP.

    " Every OTHER message the BAPI returned, as supplementary detail -
    " the generic wrapper message ("Error in document: BKPFF ...")
    " alone hides the real, specific cause (e.g. one line per rejected
    " G/L account), so nothing here is discarded, but it is no longer
    " logged as if it were its own status change.
    LOOP AT lt_return INTO DATA(ls_ret_msg).
      CHECK ls_ret_msg-message <> ls_err-message. " already logged above as the primary transition
      DATA(lv_ret_msg) = ls_ret_msg-message.
      PERFORM smart_case CHANGING lv_ret_msg.
      LOOP AT pu_chunk INTO ls_it.
        PERFORM write_log USING ls_it gc_action-bapi_detail gc_status-post_failed gc_status-post_failed
                                 lv_ret_msg.
      ENDLOOP.
    ENDLOOP.

    COMMIT WORK AND WAIT.

    " Immediate retry: as soon as the chunk becomes POST_FAILED (025),
    " try again right now, in this same run (online or background)
    " instead of waiting for the next execution. The retry is the same
    " one the next run would do - BUDAT = today (see LV_IS_RETRY at the
    " top of this FORM) - so it only makes sense when today's date was
    " NOT the one that just failed; otherwise it would fail the same
    " way. That also guarantees at most ONE extra attempt: the
    " recursive call is itself a retry (BUDAT = today), so it never
    " retries again. Log trail per item: POSTING -> POST_FAILED ->
    " POSTING -> POSTED (or POST_FAILED again, with the new message).
    IF lv_budat <> sy-datum.
      DATA lt_retry TYPE y0033fi_icm_item_tt.
      CLEAR lt_retry.
      LOOP AT pu_chunk INTO ls_it.
        ls_it-status = gc_status-post_failed.
        ls_it-msg    = ls_err-message.
        APPEND ls_it TO lt_retry.
      ENDLOOP.
      PERFORM post_one_document USING lt_retry.
    ENDIF.
    RETURN.
  ENDIF.

  CALL FUNCTION 'BAPI_TRANSACTION_COMMIT' EXPORTING wait = 'X'.

  " Confirmed against a working reference implementation in this
  " landscape: the generated document number comes back as a type S
  " message, ID 'RW', number '605', with BELNR/BUKRS/GJAHR packed into
  " MESSAGE_V2 (not MESSAGE_V1, and not GJAHR derived from BUDAT).
  "
  " CLEAR explicitly and verify before use - a real test showed the
  " PREVIOUS successful call's BELNR ending up written onto a LATER,
  " unrelated cycle's items. Inline DATA() variables are not
  " guaranteed to reset between separate calls to this FORM within
  " the same program session (which now stays alive across screens,
  " since BACK returns to the selection screen instead of exiting) -
  " if this READ TABLE ever fails to find a fresh match, we must NOT
  " silently carry over whatever value happened to be sitting there.
  CLEAR: ls_ok, lv_belnr, lv_gjahr.
  READ TABLE lt_return INTO ls_ok WITH KEY type = 'S' id = 'RW' number = '605'.
  IF sy-subrc <> 0 OR ls_ok-message_v2 IS INITIAL.
    " Could not confirm a real success message this time - do NOT
    " mark anything POSTED with a blank/guessed document number.
    " Leave the chunk locked as POSTING for manual investigation
    " rather than silently corrupting the data.
    LOOP AT pu_chunk INTO ls_it.
      PERFORM write_log USING ls_it gc_action-error gc_status-posting gc_status-posting
                               'BAPI reported success but no BELNR could be confirmed - left as POSTING for manual review'.
    ENDLOOP.
    COMMIT WORK AND WAIT.
    RETURN.
  ENDIF.
  lv_belnr = ls_ok-message_v2(10).
  lv_gjahr = ls_ok-message_v2+14(4).

  LOOP AT pu_chunk INTO ls_it.
    READ TABLE lt_buzei INTO ls_buzei WITH KEY docnum = ls_it-docnum
                                                itmnum = ls_it-itmnum
                                                seqnr  = ls_it-seqnr.
    UPDATE y0033fi_icm_item SET status   = @gc_status-posted,
                                belnr_fi = @lv_belnr,
                                gjahr_fi = @lv_gjahr,
                                buzei_fi = @ls_buzei-buzei,
                                bldat_fi = @lv_bldat,
                                budat_fi = @lv_budat,
                                period_fallback = @lv_fallback,
                                pst_user = @sy-uname,
                                pst_date = @sy-datum,
                                pst_time = @sy-uzeit,
                                msg      = @space
      WHERE docnum = @ls_it-docnum AND itmnum = @ls_it-itmnum AND seqnr = @ls_it-seqnr.
    " Clears any MSG left over from a previous failed attempt
    " (POST_FAILED) - now that this cycle succeeded, the item looks
    " like it still has a pending error otherwise. The full history
    " (including that earlier failure) stays in Y0033FI_ICM_LOG.

    ls_it-belnr_fi = lv_belnr.
    ls_it-gjahr_fi = lv_gjahr.
    ls_it-budat_fi = lv_budat.
    ls_it-bldat_fi = lv_bldat.

    PERFORM write_log USING ls_it gc_action-posted gc_status-posting gc_status-posted
                             'Accounting document posted'.
    IF lv_fallback = abap_true.
      DATA lv_msg_fallback TYPE char200.
      lv_msg_fallback = |Posting date fallback to { lv_budat DATE = USER } (period closed)|.
      PERFORM write_log USING ls_it gc_action-period_fallback gc_status-posted gc_status-posted
                               lv_msg_fallback.
    ENDIF.

    PERFORM dequeue_item USING ls_it-docnum ls_it-itmnum ls_it-seqnr.
  ENDLOOP.

  COMMIT WORK AND WAIT.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  EXECUTE_REVERSAL
*&---------------------------------------------------------------------*
* Reads items in STRQ, groups them by the original BELNR_FI/GJAHR_FI
* (one nota fiscal = one document, so this is effectively one group
* per nota), and reverses the WHOLE original document using the
* native reversal BAPI - never a hand-built inverse posting - so the
* standard SAP reversal link (STBLG/STJAH) is preserved.
*
* Items that shared that original document but are NOT part of this
* reversal request (still legitimately PST) are re-posted individually
* right after, via POST_ONE_DOCUMENT, so only the intended item(s)
* actually lose their benefit - see design note on document grouping.
*---------------------------------------------------------------------*
FORM execute_reversal.

  SELECT * FROM y0033fi_icm_item
    INTO TABLE @DATA(lt_strq)
    WHERE status = @gc_status-reversal_requested.

  " Online (screen 200 "Execute Reversal" button): only reverse what
  " the admin actually marked with SEL - same reasoning as
  " POST_RELEASED_ITEMS. SCR_200 also runs ENSURE_FULL_DOCNUM_SELECTION
  " before calling this FORM, so a partial selection within one nota's
  " items (which would leave a sibling REVERSAL_REQUESTED row stranded,
  " since a reversal always reverses the WHOLE original accounting
  " document) gets caught there first, not here.
  IF sy-batch = abap_false.
    LOOP AT lt_strq INTO DATA(ls_check).
      READ TABLE git_item WITH KEY docnum = ls_check-docnum
                                    itmnum = ls_check-itmnum
                                    seqnr  = ls_check-seqnr
        INTO DATA(ls_git_check).
      IF sy-subrc <> 0 OR ls_git_check-sel <> abap_true.
        DELETE lt_strq.
      ENDIF.
    ENDLOOP.
  ENDIF.

  IF lt_strq IS INITIAL.
    " Same reasoning as POST_RELEASED_ITEMS above - no reversal
    " requests pending (or - online - nothing was marked) is expected
    " on most runs, not an error.
    IF sy-batch = abap_true.
      MESSAGE s008(y0033icms).
    ELSE.
      MESSAGE e008(y0033icms).
    ENDIF.
    RETURN.
  ENDIF.

  SORT lt_strq BY belnr_fi gjahr_fi.

  LOOP AT lt_strq INTO DATA(ls_first) GROUP BY ( belnr = ls_first-belnr_fi
                                                  gjahr = ls_first-gjahr_fi )
       REFERENCE INTO DATA(lr_group).

    DATA(lt_req) = VALUE y0033fi_icm_item_tt( ).
    LOOP AT GROUP lr_group INTO DATA(ls_member).
      APPEND ls_member TO lt_req.
    ENDLOOP.
    READ TABLE lt_req INTO DATA(ls_any) INDEX 1.

    " Find any OTHER item still legitimately posted on the same
    " original document - these need to be re-posted after reversal.
    DATA lt_keep TYPE y0033fi_icm_item_tt.
    SELECT * FROM y0033fi_icm_item
      INTO TABLE @lt_keep
      WHERE belnr_fi = @ls_any-belnr_fi
        AND gjahr_fi = @ls_any-gjahr_fi
        AND status = @gc_status-posted.

    " OBJ_SYS must be the full logical system name (e.g. QP2CLNT600),
    " confirmed against a real document header - SY-SYSID alone only
    " gives the 3-char system ID (e.g. QP2), which is not the same
    " thing and would not match.
    DATA lv_logsys TYPE bapiache09-obj_sys.
    CALL FUNCTION 'OWN_LOGICAL_SYSTEM_GET'
      IMPORTING
        own_logical_system             = lv_logsys
      EXCEPTIONS
        own_logical_system_not_defined = 1
        OTHERS                         = 2.
    IF sy-subrc <> 0.
      lv_logsys = sy-sysid. " fallback - should not normally be needed
    ENDIF.

    DATA lv_obj_key TYPE bapiache09-obj_key.
    lv_obj_key = |{ ls_any-belnr_fi }{ ls_any-bukrs }{ ls_any-gjahr_fi }|.

    " Confirmed interface: BAPI_ACC_DOCUMENT_REV_POST (NOT
    " BAPI_ACC_DOCUMENT_REV, which does not exist and dumps with
    " FUNCTION_NOT_FOUND) - takes a REVERSAL structure (BAPIACREV)
    " plus BUS_ACT = 'RFBU', not individual EXPORTING parameters.
    " Reversal reason '01' only allows the reversal to be posted on
    " the SAME date as the original document (confirmed by a real
    " error: "Reversal reason 01 only permits posting date ..."). Use
    " the original posting date (BUDAT_FI); fall back to today only if
    " that's somehow not available.
    DATA(lv_reversal_pstng_date) = ls_any-budat_fi.
    IF lv_reversal_pstng_date IS INITIAL.
      lv_reversal_pstng_date = sy-datum.
    ENDIF.

    DATA ls_reversal TYPE bapiacrev.
    CLEAR ls_reversal.
    ls_reversal-obj_type   = 'BKPFF'.
    ls_reversal-obj_key    = lv_obj_key.
    ls_reversal-obj_key_r  = lv_obj_key.
    ls_reversal-obj_sys    = lv_logsys.
    ls_reversal-pstng_date = lv_reversal_pstng_date.
    ls_reversal-reason_rev = '01'. " TODO: confirm the correct reversal reason code with FI

    DATA lv_rev_obj_type TYPE bapiache09-obj_type.
    DATA lv_rev_obj_key  TYPE bapiache09-obj_key.
    DATA lv_rev_obj_sys  TYPE bapiache09-obj_sys.
    DATA lt_rev_return   TYPE TABLE OF bapiret2.

    " Same reasoning as POST_ONE_DOCUMENT - BKPF-TCODE otherwise
    " records SE38 for this reversal document too.
    DATA(lv_saved_tcode_rev) = sy-tcode.
    IF gv_program_id IS NOT INITIAL.
      sy-tcode = gv_program_id.
    ENDIF.

    CALL FUNCTION 'BAPI_ACC_DOCUMENT_REV_POST'
      EXPORTING
        reversal        = ls_reversal
        bus_act         = 'RFBU'
      IMPORTING
        obj_type        = lv_rev_obj_type
        obj_key         = lv_rev_obj_key
        obj_sys         = lv_rev_obj_sys
      TABLES
        return          = lt_rev_return.

    sy-tcode = lv_saved_tcode_rev. " restore immediately - never leak the fake value beyond this BAPI call

    READ TABLE lt_rev_return WITH KEY type = 'E' TRANSPORTING NO FIELDS.
    IF sy-subrc <> 0.
      READ TABLE lt_rev_return WITH KEY type = 'A' TRANSPORTING NO FIELDS.
    ENDIF.
    IF sy-subrc = 0.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      READ TABLE lt_rev_return INTO DATA(ls_rev_err) WITH KEY type = 'E'.
      IF sy-subrc <> 0.
        READ TABLE lt_rev_return INTO ls_rev_err WITH KEY type = 'A'.
      ENDIF.
      PERFORM smart_case CHANGING ls_rev_err-message.
      LOOP AT lt_req INTO DATA(ls_req).
        UPDATE y0033fi_icm_item SET status = @gc_status-posted,
                                    msg    = @ls_rev_err-message
          WHERE docnum = @ls_req-docnum AND itmnum = @ls_req-itmnum AND seqnr = @ls_req-seqnr.
        PERFORM write_log USING ls_req gc_action-error gc_status-reversal_requested gc_status-posted
                                 ls_rev_err-message.
      ENDLOOP.

      " Every OTHER message the BAPI returned, as supplementary detail
      " (not a real transition - see POST_ONE_DOCUMENT for the same
      " pattern).
      LOOP AT lt_rev_return INTO DATA(ls_rev_msg).
        CHECK ls_rev_msg-message <> ls_rev_err-message.
        DATA(lv_rev_msg) = ls_rev_msg-message.
        PERFORM smart_case CHANGING lv_rev_msg.
        LOOP AT lt_req INTO ls_req.
          PERFORM write_log USING ls_req gc_action-bapi_detail gc_status-posted gc_status-posted
                                   lv_rev_msg.
        ENDLOOP.
      ENDLOOP.

      COMMIT WORK AND WAIT.
      CONTINUE.
    ENDIF.

    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT' EXPORTING wait = 'X'.

    DATA(lv_reversal_belnr) = lv_rev_obj_key(10).

    LOOP AT lt_req INTO ls_req.
      UPDATE y0033fi_icm_item SET status          = @gc_status-reversed,
                                  reversal_exec_user  = @sy-uname,
                                  reversal_exec_date  = @sy-datum,
                                  reversal_exec_time  = @sy-uzeit,
                                  reversal_belnr      = @lv_reversal_belnr,
                                  reversal_gjahr      = @ls_any-gjahr_fi,
                                  msg                 = @space
        WHERE docnum = @ls_req-docnum AND itmnum = @ls_req-itmnum AND seqnr = @ls_req-seqnr.
      " Same reasoning as POST_ONE_DOCUMENT: clears any MSG left over
      " from an earlier failed reversal attempt on this same item - now
      " that this cycle succeeded, it shouldn't still look like it has
      " a pending error. Full history stays in Y0033FI_ICM_LOG.

      ls_req-reversal_belnr = lv_reversal_belnr.
      " WRITE_LOG's BELNR column always logs LS_ITEM-BELNR_FI (see
      " WRITE_LOG) - overwritten here with the new reversal document
      " instead of the original posted one, so this specific REVERSED
      " log row shows the document this transition actually produced
      " (same convention as POST_ONE_DOCUMENT logging the newly posted
      " document, not some prior reference). The original document is
      " still the one this whole loop is about and stays visible on
      " every earlier log row for this item (POSTED, etc.).
      ls_req-belnr_fi = lv_reversal_belnr.
      DATA lv_msg_reversed TYPE char200.
      lv_msg_reversed = |Reversed via { lv_reversal_belnr } (trigger { ls_req-reversal_trigger })|.
      PERFORM write_log USING ls_req gc_action-reversed gc_status-reversal_requested gc_status-reversed
                               lv_msg_reversed.
    ENDLOOP.
    COMMIT WORK AND WAIT.

    " Re-post the items that were still legitimately PST on the
    " reversed document, so they keep their benefit posted.
    " Each item goes back to the approval status it originally had:
    " AUTO_APPROVED (021) if its cycle was auto-approved - once POSTED,
    " Y0033FI_ICM_LOG is the only place that still records this -
    " APPROVED (020) otherwise.
    IF lt_keep IS NOT INITIAL.
      LOOP AT lt_keep ASSIGNING FIELD-SYMBOL(<ls_keep_it>).
        SELECT SINGLE @abap_true FROM y0033fi_icm_log
          INTO @DATA(lv_was_auto)
          WHERE docnum = @<ls_keep_it>-docnum
            AND itmnum = @<ls_keep_it>-itmnum
            AND seqnr  = @<ls_keep_it>-seqnr
            AND action = @gc_action-auto_approved.
        <ls_keep_it>-status = COND #( WHEN sy-subrc = 0
                                      THEN gc_status-auto_approved
                                      ELSE gc_status-approved ).
        UPDATE y0033fi_icm_item SET status = @<ls_keep_it>-status,
                                    belnr_fi = @space,
                                    gjahr_fi = @space,
                                    buzei_fi = 0
          WHERE docnum = @<ls_keep_it>-docnum AND itmnum = @<ls_keep_it>-itmnum AND seqnr = @<ls_keep_it>-seqnr.
      ENDLOOP.
      COMMIT WORK AND WAIT.
      PERFORM post_one_document USING lt_keep.
    ENDIF.

  ENDLOOP.

  " Summary message - same reasoning as POST_RELEASED_ITEMS: re-read
  " the items this run started with (LT_STRQ) and see where each one
  " landed, since the reversal/re-post logic above updates
  " Y0033FI_ICM_ITEM directly, per group, not back into this table.
  DATA lv_reversed_ok   TYPE i.
  DATA lv_reversed_fail TYPE i.
  LOOP AT lt_strq INTO DATA(ls_rev_result).
    SELECT SINGLE status FROM y0033fi_icm_item
      INTO @DATA(lv_rev_result_status)
      WHERE docnum = @ls_rev_result-docnum AND itmnum = @ls_rev_result-itmnum AND seqnr = @ls_rev_result-seqnr.
    IF lv_rev_result_status = gc_status-reversed.
      lv_reversed_ok = lv_reversed_ok + 1.
    ELSE.
      lv_reversed_fail = lv_reversed_fail + 1.
    ENDIF.
  ENDLOOP.

  IF lv_reversed_fail > 0.
    MESSAGE s023(y0033icms) WITH lv_reversed_ok lv_reversed_fail DISPLAY LIKE 'E'.
  ELSE.
    MESSAGE s023(y0033icms) WITH lv_reversed_ok lv_reversed_fail.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_LANE_FIELDCAT
*&---------------------------------------------------------------------*
FORM build_lane_fieldcat.
  DATA: ls_fcat TYPE lvc_s_fcat.

  DEFINE add_lane_field.
    CLEAR ls_fcat.
    ls_fcat-fieldname = &1.
    ls_fcat-coltext   = &2.
    ls_fcat-outputlen = &3.
    ls_fcat-edit      = &4.
    APPEND ls_fcat TO gt_fcat_lane.
  END-OF-DEFINITION.

  CLEAR gt_fcat_lane.
  add_lane_field 'BUKRS'         'Company'      6   'X'.
  add_lane_field 'VENDOR_CODE'   'Vendor Code'  12  'X'.
  add_lane_field 'VENDOR_TAX_ID' 'CNPJ/CPF'     20  'X'.
  add_lane_field 'VENDOR_REGION' 'Vendor UF'    10  'X'.
  add_lane_field 'COUNTRY'       'Country'      8   'X'.
  add_lane_field 'PLANT_CODE'    'Plant Code'   12  'X'.
  add_lane_field 'PLANT_REGION'  'Plant UF'     10  'X'.
  add_lane_field 'TRANSFER_CFOP'   'Transfer CFOP'   10 'X'.
  add_lane_field 'TRANSFER_BRANCH' 'Transfer Branch' 4  'X'.

*&---------------------------------------------------------------------*
*&      Form  CALC_MATCH_LEVEL
*&---------------------------------------------------------------------*
* MATCH_LEVEL = how many of the 9 key fields are non-blank. Called
* automatically on save - never entered by hand (see FORM
* BUILD_LANE_FIELDCAT, field is not editable in the grid).
*---------------------------------------------------------------------*
FORM calc_match_level USING ps_lane TYPE y0033fi_icm_lane
                    CHANGING pc_level TYPE n.

  DATA lv_count TYPE i.
  IF ps_lane-bukrs           IS NOT INITIAL. lv_count = lv_count + 1. ENDIF.
  IF ps_lane-vendor_code     IS NOT INITIAL. lv_count = lv_count + 1. ENDIF.
  IF ps_lane-vendor_tax_id   IS NOT INITIAL. lv_count = lv_count + 1. ENDIF.
  IF ps_lane-vendor_region   IS NOT INITIAL. lv_count = lv_count + 1. ENDIF.
  IF ps_lane-country         IS NOT INITIAL. lv_count = lv_count + 1. ENDIF.
  IF ps_lane-plant_code      IS NOT INITIAL. lv_count = lv_count + 1. ENDIF.
  IF ps_lane-plant_region    IS NOT INITIAL. lv_count = lv_count + 1. ENDIF.
  IF ps_lane-transfer_cfop   IS NOT INITIAL. lv_count = lv_count + 1. ENDIF.
  IF ps_lane-transfer_branch IS NOT INITIAL. lv_count = lv_count + 1. ENDIF.

  pc_level = lv_count.

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  SAVE_LANE_TABLE
*&---------------------------------------------------------------------*
* Recalculates MATCH_LEVEL and CHANGED_BY/CHANGED_ON for every row,
* validates VALID_FROM/VALID_TO, then replaces the whole table content
* - simple and safe for a small configuration table like this one
* (no attempt to track row-by-row deltas from the editable grid).
*---------------------------------------------------------------------*
FORM save_lane_table.

  DATA ls_lane  TYPE y0033fi_icm_lane.
  DATA lv_level TYPE n LENGTH 1.
  DATA lv_errors TYPE i.

  LOOP AT git_lane INTO ls_lane.
    IF ls_lane-valid_to < ls_lane-valid_from.
      lv_errors = lv_errors + 1.
    ENDIF.
  ENDLOOP.

  IF lv_errors > 0.
    MESSAGE e012(y0033icms) WITH lv_errors.
    RETURN.
  ENDIF.

  LOOP AT git_lane INTO ls_lane.
    DATA(lv_tabix) = sy-tabix.
    PERFORM calc_match_level USING ls_lane CHANGING lv_level.
    ls_lane-match_level = lv_level.
    ls_lane-changed_by  = sy-uname.
    ls_lane-changed_on  = sy-datum.
    MODIFY git_lane FROM ls_lane INDEX lv_tabix.
  ENDLOOP.

  DATA lt_old TYPE STANDARD TABLE OF y0033fi_icm_lane.
  SELECT * FROM y0033fi_icm_lane INTO TABLE @lt_old.
  IF lt_old IS NOT INITIAL.
    DELETE y0033fi_icm_lane FROM TABLE @lt_old.
  ENDIF.
  " DELETE ... FROM TABLE matches rows by primary key, so no WHERE
  " clause is needed - MANDT cannot be referenced explicitly in Open
  " SQL WHERE conditions (client handling is automatic).

  IF git_lane IS NOT INITIAL.
    INSERT y0033fi_icm_lane FROM TABLE git_lane.
  ENDIF.

  COMMIT WORK AND WAIT.
  MESSAGE s009(y0033icms).

ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  SAVE_PARAM_TABLE
*&---------------------------------------------------------------------*
* Replaces the whole table content, same "small config table" pattern
* as SAVE_LANE_TABLE - sets CHANGED_BY/CHANGED_ON on every save.
*---------------------------------------------------------------------*
FORM save_param_table.

  DATA ls_param TYPE y0033fi_icm_parm.

  LOOP AT git_param INTO ls_param.
    DATA(lv_tabix) = sy-tabix.
    ls_param-changed_by = sy-uname.
    ls_param-changed_on = sy-datum.
    MODIFY git_param FROM ls_param INDEX lv_tabix.
  ENDLOOP.

  DATA lt_old TYPE STANDARD TABLE OF y0033fi_icm_parm.
  SELECT * FROM y0033fi_icm_parm INTO TABLE @lt_old.
  IF lt_old IS NOT INITIAL.
    DELETE y0033fi_icm_parm FROM TABLE @lt_old.
  ENDIF.

  IF git_param IS NOT INITIAL.
    INSERT y0033fi_icm_parm FROM TABLE git_param.
  ENDIF.

  COMMIT WORK AND WAIT.
  MESSAGE s019(y0033icms).

ENDFORM.
